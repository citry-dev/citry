def content_projection(output: str, names: dict[str, str]) -> tuple[str, list[Any]]:
    """Compare application HTML and dependency payloads; graph equality is not claimed."""
    manifests = []

    def remove_graph(match: re.Match[str]) -> str:
        wire = json.loads(match[1])
        if validate_manifest(wire) is not None:
            raise AssertionError("Invalid emitted browser graph")
        manifests.append(wire)
        return ""

    output = re.sub(
        r'<script type="application/json" data-citry-graph>(.*?)</script>', remove_graph, output, flags=re.DOTALL
    )
    for wire in manifests:
        output = output.replace(wire["revision"], "normalized-graph-revision")
    output = re.sub(r"<!--citry:g1:[^>]*-->", "", output)
    output = re.sub(r' data-cid(?:-[a-z0-9]+)?(?:="[^"]*")?', "", output)
    output = re.sub(r'(?: data-citry-root)(?:="[^"]*")?', "", output)
    for render_id, name in names.items():
        output = output.replace(render_id, name)
    return output, manifests


def observe(module: Any, call: Any, variant: str) -> dict[str, Any]:
    """Measure operation counts outside timing and retain raw relationship snapshots."""
    snapshots = []
    original_snapshot = OwnershipGraph.snapshot
    original_render = runtime._render_one_traced
    calls: Counter[str] = Counter()
    candidate_counts: dict[str, int] = {}
    scheduled_functions: Counter[str] = Counter()

    def snapshot(graph: Any) -> Any:
        result = original_snapshot(graph)
        snapshots.append(result)
        return result

    def render(element: Any, parent: Any = None, provides: Any = None) -> Any:
        if type(element) is FunctionElement:
            scheduled_functions[element.comp_cls.__name__] += 1
        else:
            calls[element.comp_cls.__name__] += 1
        return original_render(element, parent, provides)

    ids._id_base = 123456
    ids._id_counter = itertools.count()
    OwnershipGraph.snapshot = snapshot
    runtime._render_one_traced = render
    try:
        with installed(module, variant, candidate_counts):
            output = call()
        generated = next(ids._id_counter)
    finally:
        OwnershipGraph.snapshot = original_snapshot
        runtime._render_one_traced = original_render
    names = {}
    counts: Counter[str] = Counter()
    for snapshot_value in snapshots:
        for row in snapshot_value.logical_instances:
            if row.render_id not in names:
                counts[row.class_id] += 1
                names[row.render_id] = f"render_{row.class_id}_{counts[row.class_id]}"
    projected, manifests = content_projection(output, names)
    return {
        "raw_html": output,
        "names": names,
        "projected_html": projected,
        "projected_digest": digest(projected),
        "bytes": len(output.encode()),
        "generated_ids": generated,
        "component_calls": dict(calls),
        "candidate_counts": candidate_counts,
        "scheduled_functions": dict(scheduled_functions),
        "snapshots": [canonical(s) for s in snapshots],
        "browser_manifests": manifests,
        "snapshot_counts": [{key: len(getattr(s, key)) for key in s.__dataclass_fields__} for s in snapshots],
        "browser_graph_counts": [
            {key: len(value) for key, value in graph.items() if isinstance(value, list)}
            for wire in manifests
            for graph in wire["graphs"]
        ],
    }


def compare(reference: dict[str, Any], candidate: dict[str, Any], variant: str) -> None:
    """Require complete application content and count actual ordinary/function executions."""
    if reference["projected_html"] != candidate["projected_html"]:
        diff = "".join(
            difflib.unified_diff(
                reference["projected_html"].splitlines(keepends=True),
                candidate["projected_html"].splitlines(keepends=True),
            )
        )
        raise AssertionError(f"Application output differs: {diff[:6000]}")
    expected = dict(reference["component_calls"])
    selected = {"Button": 114, "Icon": 40, "HeroIcon": 41} if variant != "button" else {"Button": 114}
    # Derive small/custom-case totals from actual function execution counts.
    actual = candidate["candidate_counts"]
    selected = (
        {"Button": actual.get("callbacks", 0)}
        if variant == "button"
        else {name: actual.get(name, 0) for name in selected}
    )
    for name, count in selected.items():
        remaining = expected.get(name, 0) - count
        if remaining < 0:
            raise AssertionError("Too many selected function calls")
        if remaining:
            expected[name] = remaining
        else:
            expected.pop(name, None)
    if candidate["component_calls"] != expected:
        raise AssertionError("Unexpected ordinary component execution counts")
    if reference["generated_ids"] - candidate["generated_ids"] != sum(selected.values()):
        raise AssertionError("Unexpected identity removal")
    if variant == "deferred" and candidate["scheduled_functions"] != {k: v for k, v in selected.items() if v}:
        raise AssertionError("Missing deferred function execution")
    if variant != "deferred" and candidate["scheduled_functions"]:
        raise AssertionError("Immediate variant scheduled functions")


def normalized_snapshots(row: dict[str, Any]) -> str:
    """Keep every snapshot field while normalizing only generated render ID strings."""
    text = json.dumps(row["snapshots"], sort_keys=True)
    for render_id, name in row["names"].items():
        text = text.replace(render_id, name)
    return text


def main() -> None:
    module = scenario()
    data = module.gen_render_data()
    results = []
    raw = []
    for label, inputs in (("large", data), ("one_output", {**data, "outputs": data["outputs"][:1]})):
        variants = {
            variant: observe(module, lambda inputs=inputs: module.render(inputs), variant)
            for variant in ("reference", "button", "immediate", "deferred")
        }
        expected_functions = (
            {"Button": 114, "Icon": 40, "HeroIcon": 41, "outlets": 154, "svg_replays": 32}
            if label == "large"
            else {"Button": 19, "Icon": 22, "HeroIcon": 23, "outlets": 41, "svg_replays": 17}
        )
        for mode in ("immediate", "deferred"):
            if variants[mode]["candidate_counts"] != expected_functions:
                raise AssertionError(f"Unexpected composed activation for {label}/{mode}")
        for variant in ("button", "immediate", "deferred"):
            compare(variants["reference"], variants[variant], variant)
        if normalized_snapshots(variants["immediate"]) != normalized_snapshots(variants["deferred"]):
            raise AssertionError("Immediate and deferred functions produced different ownership snapshots")
        for variant, row in variants.items():
            snapshots = row.pop("snapshots")
            row["snapshot_digests"] = [digest(snapshot) for snapshot in snapshots]
            raw.append(
                {
                    "case": label,
                    "variant": variant,
                    "snapshots": snapshots,
                    "browser_manifests": row.pop("browser_manifests"),
                }
            )
            for key in ("raw_html", "projected_html", "names"):
                row.pop(key)
        results.append({"case": label, "variants": variants})
    archive = ROOT / "benchmarks/results/repeat-render/composed-function-snapshots.json.gz"
    archive.write_bytes(gzip.compress(json.dumps(raw, separators=(",", ":")).encode(), mtime=0))
    paths = [
        Path(__file__),
        Path(__file__).with_name("adapter.py"),
        Path(__file__).with_name("plan.md"),
        ROOT / "benchmarks/wrapper_function_probe/adapter.py",
        ROOT / "benchmarks/leaf_contract_probe/adapter.py",
        ROOT / "benchmarks/template_function_probe/runtime.py",
        ROOT / "packages/py/citry/tests/test_benchmark_citry.py",
    ]
    print(
        json.dumps(
            {
                "diagnostic_only": True,
                "graph_equality_claimed_only_between_composed_variants": True,
                "cases": results,
                "snapshot_archive": {
                    "path": str(archive.relative_to(ROOT)),
                    "sha256": hashlib.sha256(archive.read_bytes()).hexdigest(),
                },
                "hashes": {str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in paths},
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    main()

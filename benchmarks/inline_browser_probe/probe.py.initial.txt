"""Check the proposed caller-owned icon behavior in installed real browsers."""

from __future__ import annotations

import argparse
import hashlib
import json
import sys
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from typing import Any

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from benchmarks.inline_browser_probe.fixtures import fixtures  # noqa: E402
from benchmarks.inline_leaf_probe.checks import installed  # noqa: E402
from benchmarks.ownership_journal_probe.probe import scenario  # noqa: E402
from playwright.sync_api import sync_playwright  # noqa: E402

READY = "window.Citry && Citry.events && Citry.events._internal.alpineStarted === true"
ICONS = ("direct", "fill", "fallback", "teleport", "conditional")


def serve(documents: dict[str, str]) -> ThreadingHTTPServer:
    """Serve self-contained documents without modifying the browser's page content API."""
    bodies = {key: value.encode() for key, value in documents.items()}

    class Handler(BaseHTTPRequestHandler):
        def do_GET(self) -> None:
            body = bodies.get(self.path)
            if body is None:
                self.send_error(404)
                return
            self.send_response(200)
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)

        def log_message(self, *_args: Any) -> None:
            pass

    server = ThreadingHTTPServer(("127.0.0.1", 0), Handler)
    threading.Thread(target=server.serve_forever, kwargs={"poll_interval": 0.01}, daemon=True).start()
    return server


def check(browser: Any, url: str, variant: str, churn: int) -> dict[str, Any]:
    """Check scope and interactions, including the intentionally changed isolation."""
    page = browser.new_page()
    errors = []
    page.on("pageerror", lambda error: errors.append(str(error)))
    page.goto(url)
    page.wait_for_function(READY)
    for name in ICONS:
        page.locator(f".{name}-icon").wait_for(state="visible")
    scopes = page.evaluate("""() => Object.fromEntries(
        ['direct','fill','fallback','teleport','conditional'].map(name => {
            const element = document.querySelector('.' + name + '-icon');
            return [name, {owner: Alpine.evaluate(element, 'typeof owner === "undefined" ? null : owner'),
                           isolated: element.hasAttribute('data-citry-root')}];
        }))""")
    expected = {
        name: {
            "owner": ("receiver" if name == "fallback" else "caller") if variant == "inline" else None,
            "isolated": variant != "inline",
        }
        for name in ICONS
    }
    if scopes != expected:
        raise AssertionError(f"Unexpected {variant} scopes: {scopes!r}")
    for count, name in enumerate(("direct", "fill", "teleport", "conditional"), 1):
        page.locator(f".{name}-icon path").click()
        page.wait_for_function(
            "expected => document.querySelector('.caller-count').textContent === String(expected)", arg=count
        )
    page.locator(".fallback-icon path").click()
    page.wait_for_function("document.querySelector('.receiver-count').textContent === '1'")
    for _ in range(churn):
        page.locator(".toggle").click()
        page.locator(".conditional-icon").wait_for(state="detached")
        page.locator(".toggle").click()
        page.locator(".conditional-icon").wait_for(state="visible")
    page.locator(".conditional-icon path").click()
    page.wait_for_function("document.querySelector('.caller-count').textContent === '5'")
    final = page.evaluate("""() => ({
        icons: document.querySelectorAll('svg').length,
        caller: document.querySelector('.caller-count').textContent,
        receiver: document.querySelector('.receiver-count').textContent,
        teleported: !!document.querySelector('#teleport-target .teleport-icon'),
        revisions: Citry.manager.ownership.revisions().length,
        conditionalOwner: Alpine.evaluate(document.querySelector('.conditional-icon'), 'typeof owner === "undefined" ? null : owner')
    })""")
    if (
        errors
        or final["icons"] != 5
        or not final["teleported"]
        or final["conditionalOwner"] != expected["conditional"]["owner"]
    ):
        raise AssertionError(f"Browser state/errors: {final!r}, {errors!r}")
    page.close()
    return {"scopes": scopes, "final": final, "page_errors": errors, "churn_cycles": churn}


def main() -> None:
    """Keep browser qualification separate from the retained server timing experiment."""
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--browsers", nargs="+", default=["chromium", "firefox", "webkit"])
    parser.add_argument("--churn", type=int, default=20)
    args = parser.parse_args()
    module = scenario()
    page_class = fixtures(module)
    documents = {}
    for variant in ("reference", "identity", "inline"):
        with installed(module, variant):
            documents["/" + variant] = page_class().render().serialize()
    server = serve(documents)
    results = {}
    try:
        with sync_playwright() as playwright:
            for name in args.browsers:
                browser = getattr(playwright, name).launch()
                try:
                    results[name] = {
                        "version": browser.version,
                        "variants": {
                            variant: check(
                                browser, f"http://127.0.0.1:{server.server_address[1]}/{variant}", variant, args.churn
                            )
                            for variant in ("reference", "identity", "inline")
                        },
                    }
                finally:
                    browser.close()
    finally:
        server.shutdown()
        server.server_close()
    paths = [
        Path(__file__),
        Path(__file__).with_name("fixtures.py"),
        ROOT / "benchmarks/inline_leaf_probe/adapter.py",
        ROOT / "benchmarks/inline_leaf_probe/checks.py",
        ROOT / "benchmarks/leaf_contract_probe/adapter.py",
    ]
    print(
        json.dumps(
            {
                "browser_qualification_only": True,
                "results": results,
                "html_hashes": {key: hashlib.sha256(value.encode()).hexdigest() for key, value in documents.items()},
                "hashes": {
                    str(path.relative_to(ROOT)): hashlib.sha256(path.read_bytes()).hexdigest() for path in paths
                },
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    main()
